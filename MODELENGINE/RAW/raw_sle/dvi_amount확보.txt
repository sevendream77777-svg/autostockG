import xml.etree.ElementTree as ET

def get_integrity_dividend(api_key, ticker_code, start_date, end_date):
    """
    무결성 배당 데이터 확보 함수
    1. Ticker -> ISIN 변환 (무결성 확보의 핵심)
    2. ISIN 기반 배당 조회 (Seibro API)
    """
    
    # 1. ISIN 조회 (getSecnInfo)
    isin_url = "http://api.seibro.or.kr/.../getSecnInfo"
    isin_params = {'serviceKey': api_key, 'shortIssuNo': ticker_code}
    
    try:
        resp = requests.get(isin_url, params=isin_params)
        tree = ET.fromstring(resp.content)
        isin = tree.find('.//issucoCustno').text
        print(f"Verified ISIN for {ticker_code}: {isin}")
    except Exception as e:
        return f"Error mapping ISIN: {e}"

    # 2. 배당금 조회 (getDivPymtInfo)
    div_url = "http://api.seibro.or.kr/.../getDivPymtInfo"
    div_params = {
        'serviceKey': api_key,
        'issucoCustno': isin,  # 12자리 코드 필수
        'stdDt': start_date,
        'edDt': end_date
    }
    
    resp_div = requests.get(div_url, params=div_params)
    div_tree = ET.fromstring(resp_div.content)
    
    results =
    for item in div_tree.findall('.//item'):
        # 사용자 스키마 매핑
        data_point = {
            'code': ticker_code,
            'date': item.find('recordDt').text,      # 기준일 -> PK
            'div_amount': item.find('cashDivAmt').text, # 핵심 데이터
            'payment_date': item.find('divPymtDt').text # 검증용 보조 데이터
        }
        results.append(data_point)
        
    return results
